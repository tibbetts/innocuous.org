{{/* OpenGraph + Twitter card metadata. Every page used to emit the site-wide description and no image at all, so every shared link previewed identically. This builds a per-page preview: text from the page's own front matter, and a card image generated at build time by drawing the title onto assets/img/og-base.png. A page can opt out of the generated card with `image:` in front matter — either a path under assets/ or an absolute URL. */}} {{/* ---- description: one chain, reused by the plain meta tag in head.html ---- */}} {{ $desc := "" }} {{ with .Description }}{{ $desc = . }} {{ else }}{{ with .Params.summary }}{{ $desc = . }} {{ else }}{{ with .Summary }}{{ $desc = . }} {{ else }}{{ $desc = .Site.Params.description }}{{ end }}{{ end }}{{ end }} {{ $desc = $desc | plainify | htmlUnescape | chomp }} {{ $desc = cond (gt (len $desc) 200) (printf "%s…" (substr $desc 0 199 | chomp)) $desc }} {{ $title := cond .IsHome .Site.Title .Title }} {{/* Section eyebrow, so a card says what kind of thing it is. */}} {{ $kind := "" }} {{ if not .IsHome }} {{ if eq .Section "projects" }}{{ $kind = "Project" }} {{ else if eq .Section "articles" }}{{ $kind = "Article" }} {{ else if eq .Section "agent-blog" }}{{ $kind = "Agent Blog" }} {{ else }}{{ $kind = .Site.Title }}{{ end }} {{ else }}{{ $kind = "Bulrush Labs" }}{{ end }} {{/* ---- the card ---- */}} {{ $ogImage := "" }} {{ $ogW := 0 }} {{ $ogH := 0 }} {{ $ogAlt := printf "%s — %s" $title .Site.Title }} {{/* Dimensions are only emitted when they are actually known: a hardcoded 1200x630 would be a lie for an overridden image, and a wrong width is worse than an absent one — scrapers lay the card out from it. */}} {{ with .Params.image }} {{ if hasPrefix . "http" }} {{ $ogImage = . }} {{ else }} {{ with resources.Get . }} {{ $ogImage = .Permalink }} {{ $ogW = .Width }} {{ $ogH = .Height }} {{ end }} {{ end }} {{ end }} {{ if not $ogImage }} {{ $base := resources.Get "img/og-base.png" }} {{ $bold := resources.Get "fonts/PlusJakartaSans-Bold.ttf" }} {{ $semi := resources.Get "fonts/PlusJakartaSans-SemiBold.ttf" }} {{ if and $base $bold $semi }} {{/* Title: 60px in a 1040px column, up to three lines. */}} {{ $lines := partial "wrap-lines.html" (dict "text" $title "budget" 24 "max" 3) }} {{ $headline := delimit $lines "\n" }} {{/* Summary underneath, positioned from the title's actual height rather than a fixed offset — the title is 1-3 lines, so a constant y would either float under short titles or collide with tall ones. Title line box is 60 + 18 linespacing = 78px from y=160. The pale band starts at y=440, so the space left over decides how many 42px summary lines fit. At three title lines nothing fits, and the summary is dropped rather than overrun the band. */}} {{ $titleLines := len $lines }} {{ $sumY := add 172 (mul $titleLines 78) }} {{ $sumMax := index (slice 0 3 2 0) $titleLines }} {{ $sumLines := partial "wrap-lines.html" (dict "text" $desc "budget" 62 "max" $sumMax) }} {{ $filters := slice (images.Text $kind (dict "font" $semi "color" "#2f7550" "size" 28 "x" 100 "y" 96)) (images.Text $headline (dict "font" $bold "color" "#161614" "size" 60 "linespacing" 18 "x" 100 "y" 160)) }} {{ with $sumLines }} {{ $filters = $filters | append (images.Text (delimit . "\n") (dict "font" $semi "color" "#4e4e48" "size" 30 "linespacing" 12 "x" 100 "y" $sumY)) }} {{ end }} {{ $filters = $filters | append (images.Text "bulrushlabs.com" (dict "font" $semi "color" "#2a6b45" "size" 30 "x" 100 "y" 512)) }} {{ $img := $base | images.Filter $filters }} {{ $ogImage = $img.Permalink }} {{ $ogW = $img.Width }} {{ $ogH = $img.Height }} {{ end }} {{ end }} {{/* The plain description lives here too, so it cannot drift from the one the preview cards use — they were already out of step, which is how every page ended up advertising the site blurb. */}} {{ with $ogImage }} {{ if and $ogW $ogH }} {{ end }} {{ end }} {{ if and (not .IsHome) (not .Date.IsZero) }} {{ with .Params.author }}{{ end }} {{ end }} {{ with $ogImage }}{{ end }}
