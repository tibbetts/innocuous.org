This builds a per-page preview: text from the page's own front matter, and a card image generated at build time by drawing the title onto assets/img/og-base.png. "http" }} {{ $ogImage = . }} {{ else }} {{ with resources.Get . }} {{ $ogImage = .Permalink }} {{ $ogW = .Width }} {{ $ogH = .Height }} {{ end }} {{ end }} {{ end }} {{ if not $ogImage }} {{ $base := resources.Get "img/og-base.png" }} {{ $bold := resources.Get "fonts/PlusJakartaSans-Bold.ttf" }} {{ $semi := resources.Get "fonts/PlusJakartaSans-SemiBold.ttf" }} {{ if and $base $bold $semi }} {{/* images.Text does not wrap, so wrap here: greedy fill to a character budget tuned to the 1040px text column at 60px, capped at three lines so a long title cannot run off the card. */}} {{ $budget := 24 }} {{ $lines := slice }} {{ $cur := "" }} {{ range (split $title " ") }} {{ $try := cond (eq $cur "") . (printf "%s %s" $cur .) }} {{ if le (len $try) $budget }} {{ $cur = $try }} {{ else }} {{ $lines = $lines | append $cur }} {{ $cur = . }} {{ end }} {{ end }} {{ if ne $cur "" }}{{ $lines = $lines | append $cur }}{{ end }} {{ if gt (len $lines) 3 }} {{ $lines = first 3 $lines }} {{ $lines = append (printf "%s…" (index $lines 2)) (first 2 $lines) }} {{ end }} {{ $headline := delimit $lines "\n" }} {{ $img := $base | images.Filter (images.Text $kind (dict "font" $semi "color" "#2f7550" "size" 28 "x" 100 "y" 96)) (images.Text $headline (dict "font" $bold "color" "#161614" "size" 60 "linespacing" 18 "x" 100 "y" 160)) (images.Text "bulrushlabs.com" (dict "font" $semi "color" "#2a6b45" "size" 30 "x" 100 "y" 512)) }} {{ $ogImage = $img.Permalink }} {{ $ogW = $img.Width }} {{ $ogH = $img.Height }} {{ end }} {{ end }} {{/* The plain description lives here too, so it cannot drift from the one the preview cards use — they were already out of step, which is how every page ended up advertising the site blurb. */}} {{ with $ogImage }} {{ if and $ogW $ogH }} {{ end }} {{ end }} {{ if and (not .IsHome) (not .Date.IsZero) }} {{ with .Params.author }}{{ end }} {{ end }} {{ with $ogImage }}{{ end }}
